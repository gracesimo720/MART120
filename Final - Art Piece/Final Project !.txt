<script>
// ------------------- VARIABLES -------------------
let circleX = 0;
let circleSpeed = 2;

// ------------------- SETUP -------------------
function setup() {
  createCanvas(600, 400);
}

// ------------------- DRAW LOOP -------------------
function draw() {
  background(220, 240, 255);

  // moving circle
  fill(100, 150, 255);
  ellipse(circleX, height / 2, 80, 80);

  // movement
  circleX += circleSpeed;

  // bounce at edges
  if (circleX > width || circleX < 0) {
    circleSpeed *= -1;
  }

  // instructions text
  fill(0);
  textSize(18);
  text("Click to change direction", 20, 30);
}

// ------------------- INTERACTION -------------------
function mousePressed() {
  circleSpeed *= -1;
}
</script>
