<script>
// ---------------- VARIABLES ----------------
let circles = [];

function setup() {
  createCanvas(windowWidth, windowHeight);
  noStroke();
  
  // Create 80 random circles
  for (let i = 0; i < 80; i++) {
    circles.push({
      x: random(width),
      y: random(height),
      size: random(20, 80),
      xSpeed: random(-1, 1),
      ySpeed: random(-1, 1)
    });
  }
}

function draw() {
  // Slightly transparent background for trailing effect
  background(10, 10, 20, 30);  
  
  for (let c of circles) {
    fill(
      150 + sin(frameCount * 0.01 + c.x) * 100,
      200 + sin(frameCount * 0.02 + c.y) * 50,
      255
    );
    
    circle(c.x, c.y, c.size);
    
    // Move circles
    c.x += c.xSpeed;
    c.y += c.ySpeed;
    
    // Bounce off edges
    if (c.x < 0 || c.x > width) c.xSpeed *= -1;
    if (c.y < 0 || c.y > height) c.ySpeed *= -1;
  }
}

// Make canvas responsive
function windowResized() {
  resizeCanvas(windowWidth, windowHeight);
}
</script>
